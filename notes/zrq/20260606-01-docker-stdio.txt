# Add STDIO streams to a Docker container.
#


  /containers/{uuid}/{stream}:
    get:
      operationId: StreamSelect
      parameters:
        - name: uuid
          in: path
          description: >-
            The UUID identifier of the container.
          required: true
          schema:
            type: string
            format: uuid
          style: simple
        - name: stream
          in: path
          description: >-
            The name of the stream [stdout|stderr].
          required: true
          schema:
            type: string
          style: simple
      responses:
        "200":
          description: >-
            200 OK - With the binary stream content.
          content:
            application/octet-stream:
              schema:
                type: string
                format: binary


    DockerContainer:
      type: object
      title: DockerContainer
      description: >-
        Details for a Docker or OCI container executable.
        See https://opencontainers.org/
      allOf:
        - $ref: '../../components.yaml#/components/schemas/AbstractExecutable'
        - type: object
          properties:
            ....
            streams:
              $ref: '#/components/schemas/DockerStdioStreams'


    DockerStdioStreams:
      description: >-
        The set of IO streams for a container.
      type: object
      title: DockerStreamSet
      properties:
        stdout:
          $ref: '#/components/schemas/DockerOutputStream'
        stderr:
          $ref: '#/components/schemas/DockerOutputStream'


    DockerOutputStream:
      description: >-
        Details of an output stream connected to the container.
      type: object
      title: DockerIOStream
      properties:
        locations:
          type: array
          items:
            type: string
            format: URL
          description: >-
            A list of URLs to access the stream.
