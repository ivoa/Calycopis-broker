# Add STDIO streams to a Docker container.
#


  /streams/{uuid}/{stream}:
    get:
      operationId: StreamSelect
      parameters:
        - name: uuid
          in: path
          description: >-
            The UUID identifier of the container.
          required: true
          schema:
            type: string
            format: uuid
          style: simple
        - name: stream
          in: path
          description: >-
            The name of the stream [stdout|stderr].
          required: true
          schema:
            type: string
          style: simple
      responses:
        "200":
          description: >-
            200 OK - With the binary stream content.
          content:
            application/octet-stream:
              schema:
                type: string
                format: binary


    DockerContainer:

# Create our platform configuration files.
#[root@calycopis-dev]

    mkdir /etc/calycopis

    cat > /etc/calycopis/database.yaml << EOF
spring:
    datasource:
        url: jdbc:postgresql://postgres:5432/calycopis
        username: $(pwgen 32 1)
        password: $(pwgen 32 1)
        driverClassName: org.postgresql.Driver
        initialize: true
EOF

    cat > /etc/calycopis/admin.yaml << EOF
calycopis:
    admin:
        username: $(pwgen 32 1)
        password: $(pwgen 32 1)
EOF
